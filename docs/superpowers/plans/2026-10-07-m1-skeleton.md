**Goal:** 搭建 vviewer monorepo 骨架：core 检测派发链、三种文件来源（本地文件/文件夹/URL）、SvelteKit 前端外壳（响应式+会话恢复）、hljs 代码查看、图片/音视频查看、CSS 变量主题框架。

```ts
  name: string;
  path: string;           // 相对 store 根的路径，'/' 分隔
  kind: Kind;
  size?: number;
  mtime?: number;         // epoch ms
}

export interface ByteRange { start: number; end: number; } // end 含

export interface TreeStore {
  readonly id: string;    // 实例唯一 id，会话恢复用
  displayName(): string;
  listChildren(path: string): Promise<TreeNode[]>;
  read(path: string, opts?: { range?: ByteRange }): Promise<Uint8Array>;

export type Encoding = 'utf-8' | 'utf-16le' | 'utf-16be' | 'gb18030';

export interface Detection {
  ext: string;            // 小写无点；无扩展名为 ''
  lang?: string;          // 代码语言（helix 名，M2 接入 languages.json）
```

```ts
// packages/core/src/detect/extension.ts
/** 从文件名/URL 提取小写扩展名；无扩展名（含点文件）返回 '' */
export function extractExtension(name: string): string {
  const clean = name.split(/[?#]/, 1)[0] ?? name;
  const base = clean.split('/').pop() ?? clean;
  const dot = base.lastIndexOf('.');
  if (dot <= 0) return ''; // <=0 覆盖无点与 .dotfile
  return base.slice(dot + 1).toLowerCase();
}
```

```ts
// packages/core/src/registry/registry.ts
import type { Renderer, RendererId } from '../types';

export class RegistryError extends Error {
  constructor(ext: string, existing: RendererId, incoming: RendererId) {
    super(`扩展名 ".${ext}" 已被 renderer "${existing}" 注册，拒绝 "${incoming}" 重复注册`);
  }
}

export interface Registry {
  install(r: Renderer): void;
  byExtension(ext: string): Renderer | undefined;
  byId(id: string): Renderer | undefined;
  all(): Renderer[];
}

export function createRegistry(): Registry {
  const byExt = new Map<string, Renderer>();
  const byId = new Map<string, Renderer>();
  return {
    install(r) {
      if (byId.has(r.id)) throw new RegistryError(`__id__`, r.id, r.id);
      for (const ext of r.extensions) {
        const owner = byExt.get(ext);
        if (owner && owner.id !== r.id) throw new RegistryError(ext, owner.id, r.id);
      }
      byId.set(r.id, r);
      for (const ext of r.extensions) byExt.set(ext, r);
    },
    byExtension: (ext) => byExt.get(ext.toLowerCase()),
    byId: (id) => byId.get(id),
    all: () => [...byId.values()]
  };
}
```

```ts
// packages/core/src/detect/magic.ts
import type { Detection } from '../types';

export function sniffSignature(head: Uint8Array): Detection['signature'] {
  const b = head;
  const starts = (sig: number[], off = 0) => sig.every((v, i) => b[off + i] === v);
  if (starts([0x50, 0x4b, 0x03, 0x04]) || starts([0x50, 0x4b, 0x05, 0x06]) || starts([0x50, 0x4b, 0x07, 0x08])) return 'zip';
  if (starts([0xd0, 0xcf, 0x11, 0xe0, 0xa1, 0xb1, 0x1a, 0xe1])) return 'ole';
  if (starts([0x89, 0x50, 0x4e, 0x47, 0x0d, 0x0a, 0x1a, 0x0a])) return 'png';
  if (starts([0xff, 0xd8, 0xff])) return 'jpeg';
  if (starts([0x47, 0x49, 0x46, 0x38])) return 'gif'; // GIF87a/GIF89a
  if (starts([0x25, 0x50, 0x44, 0x46])) return 'pdf'; // %PDF
  return null;
}
```

```ts
// packages/core/src/detect/encoding.ts
import type { Encoding } from '../types';

function hasUtf8Bom(b: Uint8Array) { return b[0] === 0xef && b[1] === 0xbb && b[2] === 0xbf; }
function hasUtf16leBom(b: Uint8Array) { return b[0] === 0xff && b[1] === 0xfe; }
function hasUtf16beBom(b: Uint8Array) { return b[0] === 0xfe && b[1] === 0xff; }

/** 严格校验 UTF-8 序列（RFC 3629，含 overlong/代理对拒绝） */
export function isStrictUtf8(b: Uint8Array): boolean {
  let i = 0;
  while (i < b.length) {
    const c = b[i]!;
    if (c < 0x80) { i += 1; continue; }
    let len: number; let min: number; let cp: number;
    if ((c & 0xe0) === 0xc0) { len = 2; min = 0x80; cp = c & 0x1f; }
    else if ((c & 0xf0) === 0xe0) { len = 3; min = 0x800; cp = c & 0x0f; }
    else if ((c & 0xf8) === 0xf0) { len = 4; min = 0x10000; cp = c & 0x07; }
    else return false;
    if (i + len > b.length) return false;
    for (let k = 1; k < len; k++) {
      const cc = b[i + k]!;
      if ((cc & 0xc0) !== 0x80) return false;
      cp = (cp << 6) | (cc & 0x3f);
    }
    if (cp < min || cp > 0x10ffff || (cp >= 0xd800 && cp <= 0xdfff)) return false;
    i += len;
  }
  return true;
}

export function detectEncoding(bytes: Uint8Array): Encoding {
  if (hasUtf8Bom(bytes)) return 'utf-8';
  if (hasUtf16leBom(bytes)) return 'utf-16le';
  if (hasUtf16beBom(bytes)) return 'utf-16be';
  if (isStrictUtf8(bytes)) return 'utf-8';
  return 'gb18030';
}
```

```ts
/** 检测链：扩展名主路由 + magic 签名 + 编码。head 建议 ≥8KB；full 用于编码精确判定。 */
export function detect(input: { name: string; head: Uint8Array; full?: Uint8Array }): Detection {
  const ext = extractExtension(input.name);
  const signature = sniffSignature(input.head);
  const sample = input.full ?? input.head;
  const encoding = detectEncoding(sample);
  const looksBinary = signature !== null || (encoding === 'gb18030' && !TEXT_EXTENSIONS.has(ext) === false && false);
  // binary 判定：图片/PDF 等签名，或 gb18030 且扩展名不在文本集合（保守）
  const gbSuspect = encoding === 'gb18030' && !TEXT_EXTENSIONS.has(ext);
  return {
    ext,
    encoding: signature !== null && signature !== 'pdf' ? undefined : encoding,
    binary: signature !== null ? !TEXT_EXTENSIONS.has(ext) : gbSuspect,
    signature,
    ...(looksBinary && false ? {} : {})
  };
}
```

```ts
// packages/core/src/dispatch/dispatcher.ts
import type { Detection, FileSource, RenderedInstance, RendererId } from '../types';
import type { Registry } from '../registry/registry';
import { detect } from '../detect';
import { errorRenderer } from './errorRenderer';

export interface Dispatcher {
  dispatch(source: FileSource, buffer: Uint8Array, target: HTMLElement):
    Promise<{ instance: RenderedInstance; rendererId: RendererId; det: Detection }>;
}

export function createDispatcher(registry: Registry): Dispatcher {
  return {
    async dispatch(source, buffer, target) {
      const head = buffer.slice(0, 8192);
      const det = detect({ name: source.name, head, full: buffer.length <= 1 << 20 ? buffer : undefined });
      let renderer = registry.byExtension(det.ext);
      try {
        if (!renderer) throw new Error(`不支持的扩展名 ".${det.ext}"`);
        let redirected = false;
        if (renderer.sniff) {
          const to = await renderer.sniff(head, det);
          if (to && to !== renderer.id) {
            const next = registry.byId(to);
            if (next) { redirected = true; renderer = next; }
          }
        }
        if (redirected && renderer.sniff) await renderer.sniff(head, det); // 允许最终 renderer 补充 sniff 元数据，但不再改派
        const instance = await renderer.render(buffer, target, source, det);
        return { instance, rendererId: renderer.id, det };
      } catch (err) {
        const message = err instanceof Error ? err.message : String(err);
        const instance = errorRenderer.show(target, message, source);
        return { instance, rendererId: errorRenderer.id, det };
      }
    }
  };
}
```

```ts
export function createUrlStore(url: string): TreeStore {
  let bytes: Promise<Uint8Array> | null = null;
  const name = decodeURIComponent(url.split('/').pop() ?? url).split(/[?#]/, 1)[0] ?? url;

  return {
    id,
    displayName: () => url,

        return new Uint8Array(await res.arrayBuffer());
      });
      return bytes;
    }
  };
}

function hash8(s: string): string {
  let h = 0x811c9dc5;
  for (let i = 0; i < s.length; i++) { h ^= s.charCodeAt(i); h = Math.imul(h, 0x01000193); }
  return (h >>> 0).toString(16).padStart(8, '0');
```

```ts
import type { TreeStore, TreeNode } from '../types';

/** 自然排序：数字段按数值比较 */
export function naturalCompare(a: string, b: string): number {
  const re = /(\d+)|(\D+)/g;
  const ax = a.match(re) ?? [];
  const bx = b.match(re) ?? [];
  for (let i = 0; i < Math.min(ax.length, bx.length); i++) {
    const an = ax[i]!, bn = bx[i]!;
    const anum = /^\d/.test(an), bnum = /^\d/.test(bn);
    if (anum && bnum) {
      const d = Number(an) - Number(bn);
      if (d !== 0) return d;
    } else if (an !== bn) {
      return an < bn ? -1 : 1;
    }
  }
  return ax.length - bx.length;
}

export function createLocalFilesStore(files: FileList | File[], label: string): TreeStore {
  const all = Array.from(files);
  const prefix = all[0]?.webkitRelativePath.split('/', 1)[0] ?? label;
  const byDir = new Map<string, File[]>();
  for (const f of all) {
    const rel = f.webkitRelativePath.split('/').slice(1).join('/'); // 去掉根目录名
    const dir = rel.split('/').slice(0, -1).join('/');
    const list = byDir.get(dir) ?? [];
    list.push(f);
    byDir.set(dir, list);
  }
  const dirIndex = new Set<string>();
  for (const [dir, list] of byDir) {
    dirIndex.add(dir);
    for (const f of list) {
      const rel = f.webkitRelativePath.split('/').slice(1).join('/');
      const segs = rel.split('/');
      for (let i = 1; i < segs.length - 1; i++) dirIndex.add(segs.slice(0, i).join('/'));
    }
  }
  return {
    id: `localfiles:${prefix}`,
    displayName: () => prefix,
    async listChildren(path) {
      const out: TreeNode[] = [];
      for (const d of dirIndex) {
        if ((d.split('/').slice(0, -1).join('/') || '') === path && d)
          out.push({ name: d.split('/').pop()!, path: d, kind: 'dir' });
      }
      for (const f of byDir.get(path) ?? []) {
        const rel = f.webkitRelativePath.split('/').slice(1).join('/');
        out.push({ name: f.name, path: rel, kind: 'file', size: f.size, mtime: f.lastModified });
      }
      return out.sort((a, b) => (a.kind === b.kind ? naturalCompare(a.name, b.name) : a.kind === 'dir' ? -1 : 1));
    },
    async read(path) {
      const f = (byDir.get(path.split('/').slice(0, -1).join('/') ?? '') ?? []).find(
        (x) => x.webkitRelativePath.split('/').slice(1).join('/') === path
      );
      if (!f) throw new Error(`未知路径: ${path}`);
      return new Uint8Array(await f.arrayBuffer());
    }
  };
```

```ts
// packages/core/src/tree/localFs.ts
import type { TreeStore, TreeNode, ByteRange } from '../types';
import { naturalCompare } from './localFiles';

type AnyHandle = FileSystemHandle & {
  kind: 'file' | 'directory';
  getFile?: () => Promise<File>;
  entries?: () => AsyncIterableIterator<[string, AnyHandle]>;
  values?: () => AsyncIterableIterator<AnyHandle>;
};

/** 请求/确认目录句柄读权限；拒绝返回 false */
export async function ensurePermission(handle: FileSystemDirectoryHandle, read = true): Promise<boolean> {
  const q = { mode: read ? 'read' : 'readwrite' } as PermissionDescriptor;
  const desc = { handle, mode: 'read' } as unknown as PermissionDescriptor & { handle: unknown };
  const query = await navigator.permissions.query(desc);
  if (query.state === 'granted') return true;
  if (query.state === 'denied') return false;
  const req = handle as FileSystemDirectoryHandle & {
    requestPermission?: (d: { mode: string }) => Promise<PermissionState>;
  };
  if (!req.requestPermission) return false;
  return (await req.requestPermission({ mode: read ? 'read' : 'readwrite' })) === 'granted';
}

export function createLocalFsStore(handle: FileSystemDirectoryHandle): TreeStore {
  async function resolve(path: string): Promise<AnyHandle> {
    let cur: AnyHandle = handle as unknown as AnyHandle;
    if (path === '') return cur;
    for (const seg of path.split('/')) {
      if (cur.kind !== 'directory') throw new Error(`路径 ${path} 中 ${seg} 的父级不是目录`);
      const next = cur.values
        ? await findValue(cur, seg)
        : await (cur as never as { getDirectoryHandle(n: string): Promise<AnyHandle>; getFileHandle(n: string): Promise<AnyHandle> })[
            undefined as never
          ];
      if (!next) throw new Error(`路径不存在: ${path}`);
      cur = next;
    }
    return cur;
  }
  async function findValue(dir: AnyHandle, name: string): Promise<AnyHandle | undefined> {
    for await (const h of (dir as unknown as FileSystemDirectoryHandle).values()) {
      if (h.name === name) return h as AnyHandle;
    }
    return undefined;
  }
  return {
    id: `localfs:${handle.name}`,
    displayName: () => handle.name,
    async listChildren(path) {
      const dir = await resolve(path);
      if (dir.kind !== 'directory') throw new Error(`${path} 不是目录`);
      const out: TreeNode[] = [];
      for await (const h of (dir as unknown as FileSystemDirectoryHandle).values()) {
        if (h.kind === 'file') {
          const f = await (h as unknown as FileSystemFileHandle).getFile();
          out.push({ name: h.name, path: path ? `${path}/${h.name}` : h.name, kind: 'file', size: f.size, mtime: f.lastModified });
        } else {
          out.push({ name: h.name, path: path ? `${path}/${h.name}` : h.name, kind: 'dir' });
        }
      }
      return out.sort((a, b) => (a.kind === b.kind ? naturalCompare(a.name, b.name) : a.kind === 'dir' ? -1 : 1));
    },
    async read(path, opts) {
      const h = await resolve(path);
      if (h.kind !== 'file' || !h.getFile) throw new Error(`${path} 不是文件`);
      const file = await h.getFile();
      if (opts?.range) {
        const { start, end } = opts.range as ByteRange;
        return new Uint8Array(await file.slice(start, end + 1).arrayBuffer());
      }
      return new Uint8Array(await file.arrayBuffer());
    }
  };
}
```

> 实现注记：`resolve` 中 `values` 分支直接用 `findValue`；不要保留 undefined-as-never 的占位分支（上面为可读性展开，实现时清理为单一逻辑）。FS Access 类型不在 TS DOM lib 全量覆盖时，允许在本文件内用最小本地 interface + `as unknown as` 收敛。

```ts
// packages/render-text/src/virtualScroller.ts
export interface VirtualScrollerHandle { destroy(): void; refresh(): void; }

/**
 * 固定行高虚拟滚动：container 内含 spacer（总高 = rows*lineHeight）与 viewport。
 * onRange(first, last) 闭区间；调用方负责填充 [first..last] 的行 DOM。
 */
export function virtualScroller(
  container: HTMLElement,
  rowCount: number,
  lineHeight: number,
  onRange: (first: number, last: number) => void,
  overscan = 10
): VirtualScrollerHandle {
  container.classList.add('vv-virtual');
  const spacer = document.createElement('div');
  spacer.className = 'vv-virtual-spacer';
  spacer.style.height = `${rowCount * lineHeight}px`;
  const viewport = document.createElement('div');
  viewport.className = 'vv-virtual-viewport';
  container.replaceChildren(spacer, viewport);
  let first = -1, last = -1;
  function update() {
    const scrollTop = container.scrollTop;
    const visible = Math.ceil(container.clientHeight / lineHeight);
    const f = Math.max(0, Math.floor(scrollTop / lineHeight) - overscan);
    const l = Math.min(rowCount - 1, f + visible + overscan * 2);
    if (f !== first || l !== last) { first = f; last = l; onRange(f, l); }
  }
  container.addEventListener('scroll', update, { passive: true });
  const ro = new ResizeObserver(update);
  ro.observe(container);
  update();
  return {
    destroy() { ro.disconnect(); container.removeEventListener('scroll', update); },
    refresh: update
  };
}
```

```ts
import type { Renderer, Encoding } from '@vviewer/core';

const MAX_HIGHLIGHT_BYTES = 5 * 1024 * 1024;
export const LINE_HEIGHT = 20;

const DECODERS: Record<Encoding, string> = {
  'utf-8': 'utf-8', 'utf-16le': 'utf-16le', 'utf-16be': 'utf-16be', 'gb18030': 'gb18030'
};

export function buildLineIndex(text: string): string[] {
  if (text === '') return [''];
  const lines = text.split('\n');
  if (lines[lines.length - 1] === '') lines.pop(); // 末尾换行不产生空行
  return lines;
}

/** 把 hljs 高亮 HTML 按行切分；跨行 span 在每行重新打开/闭合 */
export function splitHighlightedLines(html: string, lineCount: number): string[] {
  const out: string[] = new Array(lineCount).fill('');
  const stack: string[] = []; // 打开的 <span ...> 标签
  const re = /<span [^>]*>|<\/span>|\n|[^<\n]+/g;
  let line = 0;
  let m: RegExpExecArray | null;
  while ((m = re.exec(html)) !== null) {
    const tok = m[0];
    if (tok === '\n') {
      out[line] += stack.map(() => '</span>').join('');
      line += 1;
      if (line >= lineCount) break;
      out[line] += stack.join('');
      continue;
    }
    if (tok === '</span>') { stack.pop(); out[line] += tok; continue; }
    if (tok.startsWith('<span')) { stack.push(tok); out[line] += tok; continue; }
    out[line] += tok;
  }
  return out;
}

export function renderCode(
  buffer: Uint8Array,
  target: HTMLElement,
  opts: { encoding?: Encoding; highlight?: boolean } = {}
): { destroy(): void; setScrollTop(top: number): void; scrollTop(): number } {
  const enc = DECODERS[opts.encoding ?? 'utf-8'];
  const text = new TextDecoder(enc, { fatal: false }).decode(buffer);
  const lines = buildLineIndex(text);
  const doHighlight = (opts.highlight ?? true) && buffer.byteLength <= MAX_HIGHLIGHT_BYTES;
  target.classList.add('vv-code');
  const pre = document.createElement('div');
  pre.className = 'vv-code-pre';
  target.replaceChildren(pre);
  let hlLines: string[] | null = null;
  let scroller: ReturnType<typeof virtualScroller> | null = null;
  async function start() {
    if (doHighlight) {
      const hljs = await import('highlight.js');
      const { value } = hljs.highlightAuto(text);
      hlLines = splitHighlightedLines(value, lines.length);
    }
    scroller = virtualScroller(pre, lines.length, LINE_HEIGHT, (first, last) => {
      const frag = document.createDocumentFragment();
      frag.style.position = 'absolute';
      frag.style.transform = `translateY(${first * LINE_HEIGHT}px)`;
      for (let i = first; i <= last; i++) {
        const row = document.createElement('div');
        row.className = 'vv-code-line';
        row.style.height = `${LINE_HEIGHT}px`;
        const gutter = document.createElement('span');
        gutter.className = 'vv-code-gutter';
        gutter.textContent = String(i + 1);
        const body = document.createElement('span');
        body.className = 'vv-code-body';
        if (hlLines) body.innerHTML = hlLines[i] ?? '';
        else body.textContent = lines[i] ?? '';
        row.append(gutter, body);
        frag.append(row);
      }
      pre.replaceChildren(frag);
    });
  }
  void start();
  return {
    destroy() { scroller?.destroy(); pre.remove(); },
    setScrollTop(top) { pre.scrollTop = top; },
    scrollTop() { return pre.scrollTop; }
  };
}
```

```ts
/** SVG 只读消毒：去 script/on*/javascript:/外部 href|src|xlink */
export function sanitizeSvg(svgText: string): string {
  const doc = new DOMParser().parseFromString(svgText, 'image/svg+xml');
  const svg = doc.documentElement;
  if (svg.nodeName === 'parsererror' || svg.getElementsByTagName('parsererror').length) {
    throw new Error('SVG 解析失败');
  }
  for (const el of [...svg.querySelectorAll('script, foreignObject')]) el.remove();
  const walk = document.createTreeWalker(svg, NodeFilter.SHOW_ELEMENT);
  let node = walk.nextNode();
  while (node) {
    const el = node as Element;
    for (const attr of [...el.attributes]) {
      const name = attr.name.toLowerCase();
      const value = attr.value.trim().toLowerCase();
      if (name.startsWith('on')) el.removeAttribute(attr.name);
      else if ((name === 'href' || name === 'src' || name === 'xlink:href') &&
               (value.startsWith('javascript:') || value.startsWith('http:') || value.startsWith('https:') || value === '')) {
        el.removeAttribute(attr.name);
      }
    }
    node = walk.nextNode();
  }
  return new XMLSerializer().serializeToString(svg);
}

let zoom = 1;
function attachZoom(container: HTMLElement, img: HTMLElement): () => void {
  const onWheel = (e: WheelEvent) => {
    e.preventDefault();
    zoom = Math.min(10, Math.max(0.1, zoom * (e.deltaY < 0 ? 1.1 : 0.9)));
    img.style.transform = `scale(${zoom})`;
  };
  const onDbl = () => { zoom = 1; img.style.transform = 'scale(1)'; };
  // 双指捏合（pointer events 近似：两指距离比例）
  let lastDist = 0;
  const onPointerMove = (e: PointerEvent) => {
    if (!(e.buttons & 2) && e.pointerType !== 'touch') return;
    const pts = [...(container as HTMLElement & { __pts?: Set<number> }).__pts ?? []];
    void pts;
  };
  container.addEventListener('wheel', onWheel, { passive: false });
  container.addEventListener('dblclick', onDbl);
  container.addEventListener('pointermove', onPointerMove);
  return () => {
    container.removeEventListener('wheel', onWheel);
    container.removeEventListener('dblclick', onDbl);
    container.removeEventListener('pointermove', onPointerMove);
    zoom = 1;
  };
}
```

```ts
// apps/web/src/lib/stores/session.ts
export interface TabSnapshot {
  id: string;
  storeId: string;
  storeLabel: string;
  path: string;
  name: string;
  kind: 'restorable' | 'rename-only';
  scrollTop: number;
  active: boolean;
}

const DB = 'vviewer';
const STORE = 'kv';

function db(): Promise<IDBDatabase> {
  return new Promise((resolve, reject) => {
    const req = indexedDB.open(DB, 1);
    req.onupgradeneeded = () => req.result.createObjectStore(STORE);
    req.onsuccess = () => resolve(req.result);
    req.onerror = () => reject(req.error);
  });
}

async function kvGet<T>(key: string): Promise<T | null> {
  const d = await db();
  return new Promise((resolve, reject) => {
    const tx = d.transaction(STORE, 'readonly').objectStore(STORE).get(key);
    tx.onsuccess = () => resolve((tx.result as T) ?? null);
    tx.onerror = () => reject(tx.error);
  });
}

async function kvPut(key: string, value: unknown): Promise<void> {
  const d = await db();
  return new Promise((resolve, reject) => {
    const tx = d.transaction(STORE, 'readwrite').objectStore(STORE).put(value, key);
    tx.onsuccess = () => resolve();
    tx.onerror = () => reject(tx.error);
  });
}

export async function saveTabs(tabs: TabSnapshot[]): Promise<void> { await kvPut('tabs', tabs); }
export async function saveDirHandle(h: FileSystemDirectoryHandle | null): Promise<void> { await kvPut('dirHandle', h); }

export async function loadSession(): Promise<{ tabs: TabSnapshot[]; lastDirHandle: FileSystemDirectoryHandle | null }> {
  const [tabs, lastDirHandle] = await Promise.all([kvGet<TabSnapshot[]>('tabs'), kvGet<FileSystemDirectoryHandle>('dirHandle')]);
  return { tabs: tabs ?? [], lastDirHandle: lastDirHandle ?? null };
}
```

```ts
// apps/web/src/lib/openFlow.ts
import type { TreeStore, FileSource } from '@vviewer/core';
import { createSingleFileStore, createUrlStore, createLocalFsStore, createLocalFilesStore, ensurePermission } from '@vviewer/core';
import { saveDirHandle, saveTabs } from './stores/session';

export interface Tab {
  id: string;
  source: FileSource;
  scrollTop: number;
  active: boolean;
  unrestorable?: boolean;
}

let seq = 0;
const listeners = new Set<(action: { type: 'add' | 'activate' | 'close'; tab?: Tab; id?: string }) => void>();
export const tabs: Tab[] = $state([]);

export function onTabEvent(fn: (action: { type: 'add' | 'activate' | 'close'; tab?: Tab; id?: string }) => void) {
  listeners.add(fn);
  return () => listeners.delete(fn);
}
function emit(action: { type: 'add' | 'activate' | 'close'; tab?: Tab; id?: string }) {
  for (const fn of listeners) fn(action);
}

export function addTab(store: TreeStore, path: string, name: string) {
  const tab: Tab = { id: `t${++seq}`, source: { storeId: store.id, storeLabel: store.displayName(), path, name, store }, scrollTop: 0, active: true };
  for (const t of tabs) t.active = false;
  tabs.push(tab);
  emit({ type: 'add', tab });
  void persist();
  return tab;
}

export function closeTab(id: string) {
  const i = tabs.findIndex((t) => t.id === id);
  if (i >= 0) { tabs.splice(i, 1); emit({ type: 'close', id }); void persist(); }
}

export function activateTab(id: string) {
  for (const t of tabs) t.active = t.id === id;
  emit({ type: 'activate', id });
  void persist();
}

export async function persistScroll(id: string, top: number) {
  const t = tabs.find((x) => x.id === id);
  if (t) { t.scrollTop = top; await persist(); }
}

async function persist() {
  await saveTabs(tabs.map((t) => ({
    id: t.id, storeId: t.source.storeId, storeLabel: t.source.storeLabel, path: t.source.path,
    name: t.source.name,
    kind: t.source.storeId.startsWith('localfs:') ? 'restorable' : 'rename-only',
    scrollTop: t.scrollTop, active: t.active
  })));
}

export function openFiles(files: File[]) {
  for (const f of files) addTab(createSingleFileStore(f), f.name, f.name);
}

export async function openDirectoryViaPicker() {
  if (!('showDirectoryPicker' in window)) return openDirectoryViaInputFallback();
  const handle = await (window as unknown as { showDirectoryPicker(): Promise<FileSystemDirectoryHandle> }).showDirectoryPicker();
  await saveDirHandle(handle);
  addDirStoreTab(createLocalFsStore(handle));
}

export function openDirectoryViaInput(files: FileList) {
  const label = files[0]?.webkitRelativePath.split('/', 1)[0] ?? 'folder';
  addDirStoreTab(createLocalFilesStore(files, label));
}

function openDirectoryViaInputFallback() {
  const input = document.createElement('input');
  input.type = 'file';
  input.webkitdirectory = true;
  input.onchange = () => { if (input.files) openDirectoryViaInput(input.files); };
  input.click();
}

function addDirStoreTab(store: TreeStore) {
  // 目录来源以一个"目录 tab"表达：path='' name=目录名，ViewerPane 检测 kind:'dir' 时渲染为文件树
  addTab(store, '', store.displayName());
}

export async function tryRestoreDirectory(handle: FileSystemDirectoryHandle): Promise<boolean> {
  if (!(await ensurePermission(handle))) return false;
  addDirStoreTab(createLocalFsStore(handle));
  return true;
}

export function openUrl(url: string) {
  addTab(createUrlStore(url), decodeURIComponent(url.split('/').pop() ?? url), url);
}
```

```svelte
  import { createEventDispatcher } from 'svelte';

  let { store, excludedPatterns = [] }: { store: TreeStore; excludedPatterns?: string[] } = $props();
  const dispatch = createEventDispatcher<{ openfile: { path: string; name: string } }>();
  let expanded = $state<Set<string>>(new Set());
  let children = $state<Record<string, TreeNode[]>>({});
  let loading = $state<Set<string>>(new Set());

  function excluded(name: string): boolean {
    return excludedPatterns.some((p) => name === p || name.endsWith(`/${p}`) || (p.startsWith('*.') && name.endsWith(p.slice(1))));
  }

  export async function loadDir(path: string) {
    if (loading.has(path)) return;
    loading.add(path);
    try { children = { ...children, [path]: (await store.listChildren(path)).filter((n) => !excluded(n.name)) }; }
    finally { loading.delete(path); }
  }

    if (node.kind !== 'dir') { dispatch('openfile', { path: node.path, name: node.name }); return; }
    const next = new Set(expanded);
    if (next.has(node.path)) next.delete(node.path);
    else { next.add(node.path); await loadDir(node.path); }
    expanded = next;
  }
</script>
```

```svelte
  import { createRegistry, createDispatcher } from '@vviewer/core';

  import { imageRenderer, avRenderer } from '@vviewer/render-media';

  let { tab }: { tab: Tab | null } = $props();
  let host = $state<HTMLElement | null>(null);
  let scrollHost = $state<HTMLElement | null>(null);
  let instance: { destroy(): void } | null = null;
  let scrollTop = 0;

  const registry = createRegistry();
  registry.install(codeRenderer);
  registry.install(imageRenderer);
  registry.install(avRenderer);
  const dispatcher = createDispatcher(registry);

  $effect(() => {
    if (!host || !tab) return;
    if (tab.source.path === '') return; // 目录 tab 由父级渲染 FileTree
    let cancelled = false;
    void (async () => {
      instance?.destroy();
      instance = null;
      const buffer = await tab.source.store.read(tab.source.path);
      if (cancelled) return;
      const res = await dispatcher.dispatch(tab.source, buffer, host!);
      instance = res.instance;
      scrollHost = host!.closest('.vv-viewer-scroll') as HTMLElement;
    })();
    return () => { cancelled = true; instance?.destroy(); instance = null; };
  });

  function onScroll(e: Event) {
    if (!tab) return;
    scrollTop = (e.currentTarget as HTMLElement).scrollTop;
    void persistScroll(tab.id, scrollTop);
  }
</script>
```

```svelte
  import AppShell from '../lib/AppShell.svelte';

  function onDrop(e: DragEvent) {
    e.preventDefault();
    const items = e.dataTransfer?.items;
    if (items && items.length && items[0]!.kind === 'file') {
      const entry = items[0]!.webkitGetAsEntry();
      if (entry?.isDirectory) {
        // DataTransferItem 目录枚举（拖拽文件夹通道）
        const files: File[] = [];
        const reader = entry.createReader();
        const walk = (): Promise<void> => new Promise((res) =>
          reader.readEntries(async (entries) => {
            if (!entries.length) {
              openDirectoryViaInput([...files] as unknown as FileList);
              return res();
            }
            for (const en of entries) {
              const f = await (en as FileSystemFileEntry).isFile ? new Promise<File | null>((r) => (en as FileSystemFileEntry).file(r)) : null;
              if (f) files.push(Object.assign(f, { webkitRelativePath: `${entry.name}/${f!.name}` }));
            }
            await walk();
            res();
          })
        );
        void walk();
        return;
      }
    }
    if (e.dataTransfer?.files) openFiles([...e.dataTransfer.files]);
  }
```

```ts
// apps/web/e2e/m1.spec.ts
import { test, expect } from '@playwright/test';
import * as path from 'node:path';

const samples = path.resolve(__dirname, '../../../samples/m1');

test('open folder via webkitdirectory input, render code with hljs, restore session after reload', async ({ page }) => {
  await page.goto('/');
  // 通过隐藏 input 注入目录（FS Access 无法自动化，webkitdirectory 是降级通道）
  await page.evaluate(() => {
    const input = document.createElement('input');
    input.type = 'file';
    input.webkitdirectory = true;
    input.id = 'e2e-dir-input';
    input.style.display = 'none';
    document.body.appendChild(input);
    input.addEventListener('change', (e) => {
      const files = (e.currentTarget as HTMLInputElement).files;
      if (files) void (window as unknown as { __vvOpenDir: (f: FileList) => void }).__vvOpenDir(files);
    });
    (window as unknown as { __vvExposeOpenDir: () => void }).__vvExposeOpenDir = (f: FileList) => {
      void (window as unknown as { __vvOpenDirImpl?: (f: FileList) => void }).__vvOpenDirImpl?.(f);
    };
  });
  // 暴露 openDirectoryViaInput：通过 window 注入（app 内挂载调试钩子——见 Step 3）
  await page.setInputFiles('#e2e-dir-input', filesIn(samples));
  await expect(page.locator('.vv-tree')).toBeVisible();
  await expect(page.locator('.vv-tree-row', { hasText: 'hello.js' })).toBeVisible();

  await page.locator('.vv-tree-row', { hasText: 'hello.js' }).click();
  await expect(page.locator('.vv-code-pre .hljs-keyword').first()).toBeVisible();
  await expect(page.locator('.vv-tab.active', { hasText: 'hello.js' })).toBeVisible();

  // 图片 tab
  await page.locator('.vv-tree-row', { hasText: 'pixel.png' }).click();
  await expect(page.locator('.vv-image img')).toBeVisible();

  // 会话恢复（webkitdirectory 通道：刷新后 tab 为 rename-only，显示重开提示）
  await page.reload();
  await expect(page.locator('.vv-empty')).toBeVisible();
});

function filesIn(dir: string) {
  return ['hello.js', 'notes.md', 'pixel.png'].map((f) => path.join(dir, f));
}
```
